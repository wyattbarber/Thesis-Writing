\section{Introduction}

\begin{frame}{Background}
    \begin{columns}[T]        
        \begin{column}{0.49\textwidth}
            Drone Perching:
            \begin{itemize}
                \item Form of multi-modal locomotion \cite{meng_aerial_2022}.
                \item Mixes powered flight with unpowered and stationary rest \cite{meng_aerial_2022}.
            \end{itemize}
        \end{column}

        \note[item]{What is drone perching: a form of multi-modal movement in which a drone can switch between two modes:
            powered flight (the conventional drone motion), and an unpowered, stationary rest mode.}
        \note[item]{The rest mode is achieved by supporting the drones weight from a fixed object.}
        \note[item]{Example at right shows one form of perching mechanism, with a drone approching an object, perching on top and grasping,
            and later releasing and flying away.}
        
        \begin{column}{0.49\textwidth}
            \centering
            \includegraphics[width=\textwidth]{screenshots/claw.jpg}
            {\small Adapted from \cite{chi_optimized_2014}.}
        \end{column}
    \end{columns}    
\end{frame}

\begin{frame}{Significance}
    \begin{itemize}
        \item Hypothetical applications in infrastructure inspection, environmental monitoring, and disaster response
        \cite{liu_perching_2022,kirchgeorg_hedgehog_2022,hoang_autonomous_2024,kitchen_design_2020,liu_mini-drone_2023,seiber_tracking_2018}.
        \item New methods of perching may expand use of perching behaviors in real world.
    \end{itemize}

    \note[item]{Drone perching has possible use for data collection in infrastructure inspection, environmental monitoring, and 
    disaster response.}
    \note[item]{Most of the work on drone perching is in labs developing new methods, not in studies putting perching to work.}
    \note[item]{New methods of perch detection and perching that focus on cost-effectiveness and simplicity of fabrication may help 
    expand use of drone perching in real world. }
\end{frame}

\begin{frame}{Objectives}
    \begin{itemize}
        \item Design a perch detection method using low-cost and open-source components.
        \item Design a low-cost, simple to fabricate grasping device.
        \item Verify the reliability and accuracy of the design.
    \end{itemize}

    \note[item]{Design detection method, low cost and open source.}
    \note[item]{Design grasper, low cost and simple to fabricate.}
    \note[item]{Verify reliability and accuracy of both.}
\end{frame}


\section{Review of Literature}

\begin{frame}{Applications of Perching}{Infrastructure Inspection}
    \begin{columns}[T]        
        \begin{column}{0.49\textwidth}
            \begin{itemize}
                \item Drones may make inspecting power lines safer and faster \cite{chen_automatic_2018}.
                \item Perching drones could extend their mission time 
                    by perching and charging from power lines \cite{hoang_autonomous_2024,kitchen_design_2020}.
            \end{itemize}
        \end{column}        
        \begin{column}{0.49\textwidth}
            \centering
                \includegraphics[width=\textwidth]{screenshots/clasp-2.png}
                {\small Adapted from \cite{hoang_autonomous_2024}.}
        \end{column}

    \note[item]{Drones in infrastructure, specifically power lines and checking for encroaching vegetation, 
    can remove humans from situations where they are exposed to fall and electrocution hazards.}
    \note[item]{Drones can also cover long distances faster.}
    \note[item]{Perching in powerline inspection can allow drones to recharge.}
\end{frame}

\begin{frame}{Modes of Perching}
    \begin{columns}[T]        
        \begin{column}{0.49\textwidth}
            Meng et al. identified 3 primary modes of drone perching \cite{meng_aerial_2022}:
            \begin{itemize}
                \item<1-> Grasping 
                \item<2-> Embedding
                \item<3-> Attaching
            \end{itemize}
        \end{column}
        
        \begin{column}{0.49\textwidth}
            \centering
            \only<1> {
                \includegraphics[width=\textwidth]{screenshots/birdclaw-enhanced.png}
                {\small Adapted from \cite{roderick_bird-inspired_2021}.}
            }
            \only<2> {
                \includegraphics[width=\textwidth]{screenshots/fixedwing-spines.png}
                {\small Adapted from \cite{lussier_desbiens_landing_2010}.}
            }
            \only<3> {
                \includegraphics[width=\textwidth]{screenshots/electroadhesive.png}
                {\small Adapted from \cite{graule_perching_2016}.}
            }
        \end{column}
    \end{columns}    

    \note[item]{In a review of published perching techniques and devices, Meng et al identified 3
    broad category of perching mechanisms:}
    \note[item]{Grasping, often inspired by how birds perch, in which a drone mechanically grasps around an object.}
    \note[item]{Embedding, where tiny spikes grip the small scale textural features of a surface. Often used
    on vertical surfaces, such as the photo at right.}
    \note[item]{Attaching, where other means are used to adhere to a surface such as electromagnetism, adhesive compounds,
    suction cups. Tends to be surface specific, but can attach to vertical or underside of surfaces.}
\end{frame}

\begin{frame}{Methods of Perch Detection}
    \begin{columns}[T]        
        \begin{column}{0.49\textwidth}           
            \begin{itemize}
                \item<1-> Kominami et al. designed a method based on depth profiles between 
                    edges in the RGB image \cite{kominami_detection_2023}.
                \item<2-> von Frankenberg and Nokleby's method detects cylindrical shapes directly 
                    in point clouds with a RANSAC method \cite{von_frankenberg_detection_2020}.
                \item<3-> Hoang et al used a radar module paired with a camera to detect power lines \cite{hoang_autonomous_2024}.
            \end{itemize}
        \end{column}
        
        \begin{column}{0.49\textwidth}
            \centering
            \only<1> {
                \includegraphics[width=\textwidth]{screenshots/kominami-detect.png}
                {\small Adapted from \cite{kominami_detection_2023}.}
            }
            \only<2> {
                \includegraphics[width=\textwidth]{screenshots/frankenburg-detect.png}
                {\small Adapted from \cite{von_frankenberg_detection_2020}.}
            }
            \only<3> {
                \includegraphics[width=\textwidth]{screenshots/clasp-2.png}
                {\small Adapted from \cite{hoang_autonomous_2024}.}
            }
        \end{column}
    \end{columns}   

    \note[item]{Method by Kominami et al. combines Hough transform edge detection in RGB image with 
    a depth profile along a perpendicular scan line. Preliminary testing showed their method struggled with 
    low quality depth data and organic edges.}
    \note[item]{von Frankenberg and Nokelby used a method based on RANSAC detection of cylindrical object in a point cloud, 
    then evaluating continuity of detected lines and clearance aroind them to validate "perchability". They noted that this method
    did result in the detection being lost in some samples, due to RANSAC not randomly choosing enough points from the target object.}
    \note[item]{Hoang et al. also developed a detection method specifically for power lines. They used a single camera to detect orentation of 
    lines, and a radar to detect distance.}
\end{frame}

\begin{frame}{Grasper Design Goals}
    \begin{itemize}
        \item Can attach to an overhead cylindrical perch.
        \item Able to support drone of 500g with up to $63.17\frac{m}{s^2}$ acceleration
            (based on assumed sinusoid motion at 2Hz with 0.4m displacement in a tree \cite{kamimura_energy_2024}).
        \item Mass under 47.5g, or 0.33\% of the estimated payload capacity for 500g drones.
    \end{itemize}

    \note[item]{Design for overhead grasping, to tree branch or power line}
    \note[item]{Target a drone mass of 0.5kg, and an estimated worst-case dynamic load based on about 63m/s2 acceleration.}
    \note[item]{Target grasper mass of 47.5g based on 1/3 estimated payload capacity of a 500g drone.}
\end{frame}

\begin{frame}{Software Design Goals}
    \begin{itemize}
        \item Method that can work with low-cost cameras.
        \item Avoids the issues observed by von Frankenburg and Nokleby due to random sampling.
    \end{itemize}

    \note[item]{Worked with low cost cameras and open sourced image processing libraries, at the expense of lower quality data.}
    \note[item]{Chose deterministic detection methods, to avoid random sampling errors.}
\end{frame}

\begin{frame}{Software Design} 
    \begin{center}
        \includegraphics[width=0.8\textwidth]{sections/methods/sw-flow.png}\\
        {\small The proposed image processing signal flow.}  
    \end{center}
    \begin{itemize}
        \item OpenCV \cite{bradski_opencv_2000} for stereo vision.
        \item Threshold based filtering with Otsu's method \cite{otsu_threshold_1979} to remove background.
        \item 3D Hough transform designed by Dalitz et al. \cite{dalitz_iterative_2017} to detect lines.
        \item Closest line to the camera is selected as the most likely perch.
    \end{itemize}

    \note{High-level software architecture shown above. Key steps:}
    \note[item]{Point clouds are generated from grayscale images with OpenCV library}
    \note[item]{Otsu's thresholding is used to remove light colored background pixels}
    \note[item]{3D Hough transform from Dalitz et al. used to identify lines}
    \note[item]{Closest line to the camera is selected as the most likely perch}
\end{frame}

\section{Discussion}

\begin{frame}{Grasper}{Strength and Repeatability}
    \begin{itemize}
        \item Verified holding capacity is over the target, could support 
            at least 1.17kg with dynamic load. 
        \item Controller demonstrates sufficient reliability for perching and releasing. 
    \end{itemize}
\end{frame}

\begin{frame}{Grasper}{Weight}
    \begin{itemize}
        \item Weight of a single grasper is under the target, but whole device is well over.
        \item Strength is also over the target.
        \item Could be used for larger drone with more payload, or size and weight could be reduced sacrificing strength.
    \end{itemize}
\end{frame}

\begin{frame}{Perch Detection}{Impact of Distance}
    \begin{itemize}
        \item For distance and elevation measurements, mean and variance of errors decrease as distance decreases.
        \item 0 standard deviation at 2m is a result of stereo matching quantization.
        \item Azimuth errors show increasing variance at smaller distances.
        \item Increasing variance could pose problems for feedback controls as drone approaches a perch. 
    \end{itemize}
\end{frame}

\begin{frame}{Perch Detection}{Impact of Background}
    \begin{itemize}
        \item Background influences stereo matching quality and background filtering success, 
            with more complicated backgrounds producing lower detection success rates.
        \item Fine and repeating patterns seem to produce low-quality stereo disparity.
        \item Dark backgrounds prevent background removal from eliminating low-quality pixels. 
    \end{itemize}
\end{frame}

\begin{frame}{Perch Detection}{Line Identification}
    \begin{columns}[T]        
        \begin{column}{0.49\textwidth}
            \begin{itemize}
                \item 3D Hough transform \cite{dalitz_iterative_2017} struggles to separate lines from non-lines.
                \item Fixing this would reduce sensitivity to low-quality background points.
                \item Adjusting the Hough transform to consider uniqueness of a fit may help.
            \end{itemize}           
        \end{column}
        \begin{column}{0.49\textwidth}
            \centering
            \includegraphics[width=\textwidth]{sections/methods/nothresh-ptcloud.png}
            {Point cloud showing false line detections.}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Perch Detection}{Perch Shapes}
    \begin{itemize}
        \item Detection was only tested with straight, unbranching objects.
        \item May fail to detect an object that is curved or branched but still suitable 
            to grasp.
        \item To grasp, the drone would need to approach close enough to only see an unbranched 
            segment of the perch.
    \end{itemize}
\end{frame}

\begin{frame}{Perch Detection}{Field of View}
    \begin{itemize}
        \item Field of view is limited, even after adjusting camera baseline.
        \item Custom carrier PCB could help place cameras closer, and may reduce weight.
        \item Multiple stereo matching passes with different parameters could also improve field of view.
    \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
    \begin{itemize}
        \item The grasper design is successful in function, and is simple to adapt to different drone designs.
        \item Perch detection method shows promise under ideal conditions.
        \item Future work needed to improve perch detection resilience to varying background scenes, and improve 
            stability of perch orientation estimation.
    \end{itemize}
\end{frame}
